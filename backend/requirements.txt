# PaperLens processing service dependencies.
#
# Versions are pinned exactly (D-029). The previous file pinned nothing at all,
# so two installs a month apart could resolve different versions and builds were
# not reproducible (AUDIT.md 4.9).
#
# Two groups, marked below by how much confidence each pin carries:
#
#   [verified]        The version the 147-test backend suite actually passed
#                     against on this machine.
#   [not installed]   Resolved from PyPI but NOT exercised here, because the
#                     package is absent from the local environment. Re-verify
#                     after a clean install.
#
# `python-jose` is deliberately absent. It was listed before but no code path used
# it, and D-019/D-021 retire end-user JWT verification entirely: this service
# authenticates *callers* with a shared service token, never users. Keeping a JWT
# library would imply a protection that does not exist.

# --- Web framework -----------------------------------------------------------
fastapi==0.138.1          # [verified]
uvicorn==0.49.0           # [verified]

# Required by FastAPI at decorator-evaluation time for Form/File parameters. No
# route uses them now -- the browser uploads straight to Supabase Storage
# (D-016) -- but FastAPI imports it eagerly, and its absence was one of the
# failures that made the old service unimportable (AUDIT.md 3.2).
python-multipart==0.0.32  # [not installed]

# --- PDF and OCR -------------------------------------------------------------
pymupdf==1.28.0           # [verified]
pillow==12.2.0            # [verified] renders pages to images for OCR
pytesseract==0.3.13       # [not installed] also needs the Tesseract binary

# --- Analysis ----------------------------------------------------------------
# Advisory similarity only (D-024). The pipeline degrades gracefully without it:
# repeat counts, weightage and priority all come from exact hashing.
rapidfuzz==3.14.5         # [not installed]

# --- Database ----------------------------------------------------------------
supabase==2.31.0          # [not installed]

# --- Tests -------------------------------------------------------------------
pytest==9.1.1             # [verified]
httpx==0.28.1             # [verified] required by fastapi.testclient
